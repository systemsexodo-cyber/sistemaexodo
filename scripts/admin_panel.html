// Migration System
async function listMigrations() {
  const res = await fetch('/admin/migrations/list', {method: 'GET', headers: {'Accept': 'application/json'}});
  const data = await res.json();
  let html = '
Migrations de Schema (' + data.migrations.length + ' total)
    ';
  data.migrations.forEach(m => {
    const cls = m.aplicada ? 'success' : 'warning';
    html += '
  • ' + m.versao + ' - ' + m.arquivo + ' ' +
      (m.aplicada ? '(aplicada)' : '(pendente)') + '
    • ';
  });
  html += '
';
  showToast('Lista carregada', 'info');
  document.getElementById('migracaoInfo').innerHTML = html;
}

async function runMigration() {
  const versao = prompt('Versao da migration (ex: 001):');
  if (!versao) return;
  showToast('Aplicando migration ' + versao + '...', 'info');
  const res = await fetch('/admin/migrations/run', {
    method: 'POST',
    headers: {'Content-Type': 'application/json'},
    body: JSON.stringify({versao: versao, descricao: 'Admin'})
  });
  const d = await res.json();
  if (d.sucesso) {
    showToast('Migration ' + versao + ' aplicada!', 'success');
    listMigrations();
  } else {
    showToast('Erro: ' + (d.erro || ''), 'error');
  }
}

// Auto-inicializar ao carregar
document.addEventListener('DOMContentLoaded', () => {
  const md = document.createElement('div');
  md.innerHTML = '
Versao Schema:-Pending:-
';
  const card = document.querySelector('.card:first-child');
  if (card) card.querySelector('.card-body').insertBefore(md, card.firstChild);
  listMigrations();
});
